## JavaScript (Regex + Template Literals)

```js
const routePattern = /^(?<method>GET|POST|PATCH)\s+(?<path>\/api\/[\w/-]+)$/i;

export function parseRoute(input) {
  const match = routePattern.exec(input?.trim() || '');
  if (!match?.groups) throw new Error(`Invalid route: ${input}`);

  const { method, path } = match.groups;
  return {
    method,
    path,
    cacheKey: `${method}:${path}`,
  };
}
```

## Line Highlight + Filename Demo

```ts src/lib/route-parser.ts {1,6-10}
type ParsedRoute = {
  method: 'GET' | 'POST' | 'PATCH';
  path: string;
  cacheKey: string;
};

export function toCacheKey(input: ParsedRoute): string {
  const normalizedPath = input.path.replace(/\/+$/, '');
  return `${input.method}:${normalizedPath}`;
}
```

## SQL (CTE + JSON)

```sql
WITH active_users AS (
  SELECT id, email, metadata
  FROM users
  WHERE active = true
),
user_stats AS (
  SELECT
    id,
    email,
    metadata->>'plan' AS plan,
    jsonb_array_length(COALESCE(metadata->'roles', '[]'::jsonb)) AS role_count
  FROM active_users
)
SELECT *
FROM user_stats
WHERE role_count > 0
ORDER BY email ASC;
```

## Rust + Lifetimes

```rust
fn pick_longer<'a>(left: &'a str, right: &'a str) -> &'a str {
    if left.len() >= right.len() { left } else { right }
}

fn main() {
    let a = String::from("docline");
    let b = "docs";
    println!("{}", pick_longer(&a, b));
}
```
