## TypeScript

```ts
export type User = {
  id: string;
  email: string;
  roles: string[];
  active: boolean;
};

export function canEdit(user: User): boolean {
  return user.active && user.roles.includes('editor');
}
```

## Python

```python
from dataclasses import dataclass

@dataclass
class Job:
    id: str
    retries: int = 0


def run(job: Job) -> str:
    return f"running {job.id} (retries={job.retries})"
```
