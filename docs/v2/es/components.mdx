## Bloques API

<Endpoint method="POST" path="/v1/documents/import">
	Crea una nueva importacion de documentos.
</Endpoint>

<ParamTable>
	<Param name="sourceUrl" type="string" required={true}>URL publica del contenido.</Param>
	<Param name="language" type="string" required={false}>Idioma preferido (ejemplo: es, en).</Param>
	<Param name="overwrite" type="boolean" required={false}>Reemplaza el contenido existente.</Param>
</ParamTable>

<ResponseExample status="201" title="Respuesta de ejemplo">

	"createdAt": "2026-08-01T11:20:00Z"
}`}
